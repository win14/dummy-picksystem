INVOICE

PATH : /api/v1/invoice/process
HEADER : -
BODY :
    - pickerName (M) String : nama dari picker

CONTOH 

curl --request GET \
  --url http://localhost:3000/api/v1/invoice/process \
  --header 'Accept: */*' \
  --header 'Accept-Encoding: gzip, deflate, br' \
  --header 'Connection: keep-alive' \
  --header 'Content-Type: application/json' \
  --header 'User-Agent: EchoapiRuntime/1.1.0' \
  --data '{
    "invoiceID" : "2026074289",
    "pickerName" : "puji"
}'
